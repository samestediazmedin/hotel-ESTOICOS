    import { io, Socket } from 'socket.io-client';

    let sharedSocket: Socket | null = null;
    let currentToken: string | null = null;

    export function getOrCreateSocket(accessToken: string): Socket {
      if (sharedSocket && currentToken === accessToken) return sharedSocket;
      if (sharedSocket) { sharedSocket.disconnect(); sharedSocket = null; }
      const apiUrl = import.meta.env.VITE_API_URL ?? '';
      sharedSocket = io(apiUrl, {
        auth: { token: accessToken },
        reconnection: true,
        reconnectionDelay: 1000,
        reconnectionAttempts: Infinity,
        transports: ['websocket', 'polling'],
        withCredentials: true,
      });
      currentToken = accessToken;
      return sharedSocket;
    }

    export function disconnectSocket(): void {
      sharedSocket?.disconnect();
      sharedSocket = null;
      currentToken = null;
    }
    ```

    import type { GuestContactEventDto, ContactMethod } from './types';

    export interface CreateContactEventInput {
      method: ContactMethod;
      notes?: string;
    }

    export async function createContactEvent(
      guestId: string,
      input: CreateContactEventInput,
    ): Promise<GuestContactEventDto> {
      const res = await api.post<GuestContactEventDto>(
        `/guests/${guestId}/contact-events`,
        input,
      );
      return res.data;
    }

    export async function listContactEvents(
      guestId: string,
      limit = 5,
    ): Promise<GuestContactEventDto[]> {
      const res = await api.get<GuestContactEventDto[]>(
        `/guests/${guestId}/contact-events`,
        { params: { limit } },
      );
      return res.data;
    }
    ```
