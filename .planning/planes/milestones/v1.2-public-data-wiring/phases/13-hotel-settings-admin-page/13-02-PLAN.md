    **A) Implement `hotel-photos.service.ts`:**

    ```typescript
    import {
      BadRequestException,
      Injectable,
      Logger,
      NotFoundException,
      UnprocessableEntityException,
    } from '@nestjs/common';
    import { ConfigService } from '@nestjs/config';
    import {
      DeleteObjectCommand,
      HeadObjectCommand,
      PutObjectCommand,
      S3Client,
    } from '@aws-sdk/client-s3';
    import { getSignedUrl } from '@aws-sdk/s3-request-presigner';

    import { PrismaService } from '../../prisma/prisma.service';
    import { createR2Client } from '../inventory/photos/r2.client';
    import type { PresignHotelPhotoDto } from './dto/presign-hotel-photo.dto';
    import type { ConfirmHotelPhotoDto } from './dto/confirm-hotel-photo.dto';

    const MAX_SIZE_BYTES = 5 * 1024 * 1024; // 5MB — gallery UX constraint
    const PRESIGN_EXPIRY_SECONDS = 300; // 5 min

    @Injectable()
    export class HotelPhotosService {
      private readonly logger = new Logger(HotelPhotosService.name);
      private readonly r2Client: S3Client;
      private readonly bucket: string;
      private readonly publicUrl: string;

      constructor(
        private readonly prisma: PrismaService,
        private readonly config: ConfigService,
      ) {
        const accountId = this.config.get<string>('R2_ACCOUNT_ID');
        const accessKeyId = this.config.get<string>('R2_ACCESS_KEY_ID');
        const secretAccessKey = this.config.get<string>('R2_SECRET_ACCESS_KEY');
        const bucket = this.config.get<string>('R2_BUCKET');
        const publicUrl = this.config.get<string>('R2_PUBLIC_URL');

        if (!accountId) throw new Error('Missing env var: R2_ACCOUNT_ID');
        if (!accessKeyId) throw new Error('Missing env var: R2_ACCESS_KEY_ID');
        if (!secretAccessKey) throw new Error('Missing env var: R2_SECRET_ACCESS_KEY');
        if (!bucket) throw new Error('Missing env var: R2_BUCKET');
        if (!publicUrl) throw new Error('Missing env var: R2_PUBLIC_URL');

        this.bucket = bucket;
        this.publicUrl = publicUrl.replace(/\/$/, '');
        this.r2Client = createR2Client({ accountId, accessKeyId, secretAccessKey });
      }

      async presignUpload(
        dto: PresignHotelPhotoDto,
      ): Promise<{ uploadUrl: string; key: string; expiresIn: number }> {
        if (dto.size > MAX_SIZE_BYTES) {
          throw new BadRequestException('File exceeds 5MB limit');
        }
        const sanitized = dto.filename.replace(/[^a-zA-Z0-9._-]/g, '_');
        const key = `hotel-photos/${Date.now()}-${sanitized}`;
        const command = new PutObjectCommand({
          Bucket: this.bucket,
          Key: key,
          ContentType: dto.contentType,
        });
        const uploadUrl = await getSignedUrl(this.r2Client, command, {
          expiresIn: PRESIGN_EXPIRY_SECONDS,
        });
        return { uploadUrl, key, expiresIn: PRESIGN_EXPIRY_SECONDS };
      }

      async confirmUpload(
        dto: ConfirmHotelPhotoDto,
      ): Promise<{ id: string; url: string; alt: string; displayOrder: number }> {
        if (!dto.key.startsWith('hotel-photos/')) {
          throw new BadRequestException('Key must start with hotel-photos/');
        }

        try {
          await this.r2Client.send(
            new HeadObjectCommand({ Bucket: this.bucket, Key: dto.key }),
          );
        } catch {
          throw new UnprocessableEntityException(
            'R2 object not found — upload may have failed or key is invalid',
          );
        }

        const existing = await this.prisma.hotelPhoto.findFirst({ where: { key: dto.key } });
        if (existing) {
          return {
            id: existing.id,
            url: `${this.publicUrl}/${existing.key}`,
            alt: existing.alt,
            displayOrder: existing.displayOrder,
          };
        }

        const maxRow = await this.prisma.hotelPhoto.aggregate({
          _max: { displayOrder: true },
        });
        const nextOrder = (maxRow._max.displayOrder ?? -1) + 1;

        const photo = await this.prisma.hotelPhoto.create({
          data: {
            url: '', // intentional: URL resolved at read time via key (Anti-Pattern #4)
            key: dto.key,
            alt: dto.alt ?? '',
            displayOrder: nextOrder,
          },
        });

        return {
          id: photo.id,
          url: `${this.publicUrl}/${photo.key}`,
          alt: photo.alt,
          displayOrder: photo.displayOrder,
        };
      }

      async reorderPhotos(photoIds: string[]): Promise<void> {
        await this.prisma.$transaction(
          photoIds.map((id, index) =>
            this.prisma.hotelPhoto.update({
              where: { id },
              data: { displayOrder: index },
            }),
          ),
        );
      }

      async deletePhoto(id: string): Promise<void> {
        const photo = await this.prisma.hotelPhoto.findUnique({ where: { id } });
        if (!photo) throw new NotFoundException(`HotelPhoto ${id} not found`);

        if (photo.key) {
          try {
            await this.r2Client.send(
              new DeleteObjectCommand({ Bucket: this.bucket, Key: photo.key }),
            );
          } catch (err) {
            // Best-effort — log but do not block DB delete (CONTEXT.md decision)
            this.logger.warn(
              `Failed to delete R2 object ${photo.key}: ${(err as Error).message}`,
            );
          }
        }

        await this.prisma.hotelPhoto.delete({ where: { id } });
      }
    }
    ```

<threat_model>
- **Auth bypass:** Class-level `@UseGuards(JwtAuthGuard, RolesGuard)` + method-level `@Roles('ADMIN')` enforced on all 4 endpoints. Verified manually with non-admin JWT.
- **R2 path traversal:** Confirm DTO requires `key.startsWith('hotel-photos/')` via Zod regex AND service-level check (defense in depth). An admin could theoretically upload with arbitrary key via direct presign-key forgery, but ADMIN role is high-trust; abuse here is a personnel issue, not a security boundary.
- **Presigned URL hijack:** Presigned URLs expire in 300 seconds (5 min). If intercepted, the window is small. CORS on the R2 bucket already constrains origins (Phase 2 set this up).
- **Race condition on reorder:** `$transaction` ensures all `displayOrder` updates apply atomically — partial reorder cannot happen even under concurrent admin edits (the second transaction blocks on the same rows or is overridden cleanly).
- **R2 delete orphan:** If R2 delete fails, the DB row is still deleted (logged warn). Orphan R2 files are acceptable (cost is negligible at this scale; cleanup job deferred to v1.3 per CONTEXT.md).
