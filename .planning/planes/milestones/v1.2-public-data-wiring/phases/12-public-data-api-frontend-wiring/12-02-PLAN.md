    ```typescript
    import { Injectable } from '@nestjs/common';
    import { PrismaClient } from '@prisma/client';
    import { SystemConfigService } from '../../system-config/system-config.service';
    import type { PublicHotelInfoDto } from './dto/public-hotel-info.dto';
    import type { PublicRoomTypeDto } from './dto/public-room-type.dto';
    import type { PublicHotelPhotoDto } from './dto/public-hotel-photo.dto';

    // v1.2 hardcoded constants — real reviews come in Phase 14
    const RATING_PLACEHOLDER = 4.84;
    const REVIEW_COUNT_PLACEHOLDER = 318;

    // No address column exists in system_config (Phase 13 will add it). For Phase 12, return constant.
    const HOTEL_ADDRESS_PLACEHOLDER = 'La Candelaria, Bogotá';

    @Injectable()
    export class PublicPortalService {
      // Inject PrismaService if it exists in the project, otherwise instantiate PrismaClient.
      // Read existing modules (e.g., InventoryRepository) to confirm the project pattern — use the SAME pattern.
      constructor(
        private readonly prisma: PrismaClient,  // or PrismaService — match project pattern
        private readonly systemConfigService: SystemConfigService,
      ) {}

      async getHotelInfo(): Promise<PublicHotelInfoDto> {
        const config = await this.systemConfigService.getConfig();
        // If getConfig() does not exist, use this.prisma.systemConfig.findFirst()
        return {
          name: config?.hotelName ?? 'Hotel Sumapaz',
          address: HOTEL_ADDRESS_PLACEHOLDER,
          tagline: config?.tagline ?? '',
          description: config?.description ?? '',
          phone: config?.phone ?? '',
          rating: RATING_PLACEHOLDER,
          reviewCount: REVIEW_COUNT_PLACEHOLDER,
          tags: config?.tags ?? [],
        };
      }

      async getPublishedRoomTypes(): Promise<PublicRoomTypeDto[]> {
        const r2PublicUrl = process.env.R2_PUBLIC_URL ?? '';
        const roomTypes = await this.prisma.roomType.findMany({
          where: { isPublished: true, isActive: true },
          orderBy: { basePrice: 'asc' },
          include: {
            rooms: {
              where: { isActive: true },
              take: 1,
              include: {
                photos: { orderBy: { order: 'asc' }, take: 3 },
              },
            },
          },
        });

        return roomTypes.map((rt, idx) => {
          const firstRoom = rt.rooms[0];
          const photos = (firstRoom?.photos ?? []).map((p) => ({
            url: `${r2PublicUrl}/${p.key}`,
            alt: rt.name,
          }));
          const badge =
            idx === 0 ? ('Más económica' as const)
            : idx === 1 ? ('Mejor valor' as const)
            : null;
          return {
            id: rt.id,
            name: rt.name,
            capacity: rt.maxOccupancy,
            description: rt.description ?? '',
            basePrice: Number(rt.basePrice),  // Decimal → number (Pitfall #3)
            photos,
            badge,
          };
        });
      }

      async getHotelPhotos(): Promise<PublicHotelPhotoDto[]> {
        const photos = await this.prisma.hotelPhoto.findMany({
          orderBy: { displayOrder: 'asc' },
        });
        return photos.map((p) => ({
          url: p.url,
          alt: p.alt,
          displayOrder: p.displayOrder,
        }));
      }
    }
    ```

    ```typescript
    import { Controller, Get, Header } from '@nestjs/common';
    import { PublicPortalService } from './public-portal.service';
    import type { PublicHotelInfoDto } from './dto/public-hotel-info.dto';
    import type { PublicRoomTypeDto } from './dto/public-room-type.dto';
    import type { PublicHotelPhotoDto } from './dto/public-hotel-photo.dto';

    /**
     * PublicPortalController — public-facing hotel marketing data (Phase 12).
     *
     * NO @UseGuards() — JwtAuthGuard is NOT global in this app.
     * NO CSRF — read-only GETs.
     * NO ThrottlerGuard — read-only marketing data, no abuse vector.
     *
     * Shares the `/api/public` prefix with PublicBookingController. NestJS routes by
     * full method+path; no collision.
     *
     * Cache-Control: public, max-age=60, s-maxage=60 enables CDN edge caching with a
     * 60-second invalidation window — admin changes propagate within ~1 minute.
     */
    @Controller('public')
    export class PublicPortalController {
      constructor(private readonly service: PublicPortalService) {}

      @Get('hotel-info')
      @Header('Cache-Control', 'public, max-age=60, s-maxage=60')
      async getHotelInfo(): Promise<PublicHotelInfoDto> {
        return this.service.getHotelInfo();
      }

      @Get('room-types')
      @Header('Cache-Control', 'public, max-age=60, s-maxage=60')
      async getRoomTypes(): Promise<PublicRoomTypeDto[]> {
        return this.service.getPublishedRoomTypes();
      }

      @Get('hotel-photos')
      @Header('Cache-Control', 'public, max-age=60, s-maxage=60')
      async getHotelPhotos(): Promise<PublicHotelPhotoDto[]> {
        return this.service.getHotelPhotos();
      }
    }
    ```
