    ```tsx
    // ... existing imports
    import { useQueryClient } from '@tanstack/react-query';
    import { useHotelInfo } from './hooks/useHotelInfo';
    import { useRoomTypes } from './hooks/useRoomTypes';
    import { useHotelPhotos } from './hooks/useHotelPhotos';
    import { HOTEL_INFO_FALLBACK } from './data/hotel';
    import {
      HeroGallerySkeleton,
      HotelIdentitySkeleton,
      RoomsSectionSkeleton,
    } from './components/skeletons';

    export function HotelHomePage() {
      useForceLightTheme();
      const queryClient = useQueryClient();

      const infoQuery = useHotelInfo();
      const photosQuery = useHotelPhotos();
      const roomsQuery = useRoomTypes();

      // data is always defined (placeholderData) but TS may type as undefined — coerce safely
      const hotelInfo = infoQuery.data ?? HOTEL_INFO_FALLBACK;
      const photos = photosQuery.data ?? [];
      const rooms = roomsQuery.data ?? [];

      const anyError = infoQuery.isError || photosQuery.isError || roomsQuery.isError;
      const retryAll = () => {
        queryClient.invalidateQueries({ queryKey: ['public'] });
      };

      return (
        <div className="hos min-h-screen bg-warm-white text-ink-1 font-body flex flex-col">
          <TopNav hotelName={hotelInfo.hotelName} /* other props as before */ />
          <main className="flex-1">
            {anyError && (
              <div
                role="alert"
                data-testid="portal-error-banner"
                className="mx-4 lg:mx-auto lg:max-w-6xl mt-4 flex items-center gap-3 p-4 rounded-xl bg-terracotta-tint border border-terracotta text-sm text-terracotta-deep"
              >
                <span className="flex-1">No pudimos cargar parte de la información. Intentar de nuevo.</span>
                <button
                  type="button"
                  onClick={retryAll}
                  className="underline font-medium hover:text-terracotta"
                >
                  Reintentar
                </button>
              </div>
            )}

            <section id="inicio">
              {photosQuery.isPending ? (
                <HeroGallerySkeleton />
              ) : (
                <HeroGallery photos={photos} />
              )}
              {infoQuery.isPending ? (
                <HotelIdentitySkeleton />
              ) : (
                <HotelIdentity hotelInfo={hotelInfo} />
              )}
            </section>

            <section id="habitaciones">
              {roomsQuery.isPending ? (
                <RoomsSectionSkeleton />
              ) : (
                <RoomsSection rooms={rooms} />
              )}
            </section>

            {/* OTHER existing sections (ConciergeTeaser, RestaurantSection, LocationSection, ReviewsSection) — preserve verbatim. They don't depend on Phase 12 hooks. */}
          </main>
          <PortalFooter hotelInfo={hotelInfo} />
        </div>
      );
    }
    ```
