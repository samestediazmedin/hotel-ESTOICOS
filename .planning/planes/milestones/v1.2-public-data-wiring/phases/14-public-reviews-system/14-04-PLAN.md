    3. **Create `apps/web/src/features/public-portal/hooks/useReviews.ts`** — paginated with append semantics. Two approaches — pick (a) for simplicity:

       **Approach (a) — Manual page state (recommended, simpler):**
       ```typescript
       import { useState } from 'react';
       import { useQuery } from '@tanstack/react-query';
       import { fetchPublicReviews } from '../api/public-portal.api';
       import type { Review, PublicReviewsResponse } from '../types';

       const LIMIT = 10;

       export function useReviews() {
         const [page, setPage] = useState(1);

         const query = useQuery<PublicReviewsResponse, Error>({
           queryKey: ['public', 'reviews', { page, limit: LIMIT }],
           queryFn: () => fetchPublicReviews(page, LIMIT),
           staleTime: 60_000,
         });

         // Append semantics: we keep an accumulated array of all loaded reviews
         const [loaded, setLoaded] = useState<Review[]>([]);

         // Append fresh page into loaded list when query data changes
         // (naive: in real implementation guard against double-fetch via useEffect comparing page)
         // For simplicity, expose query data + loadMore that increments page.

         const hasMore = query.data ? page < query.data.pages : false;
         const loadMore = () => setPage((p) => p + 1);

         return {
           reviews: query.data?.reviews ?? [],
           total: query.data?.total ?? 0,
           averageRating: query.data?.averageRating ?? 0,
           pages: query.data?.pages ?? 0,
           isLoading: query.isLoading,
           isError: query.isError,
           hasMore,
           loadMore,
           page,
         };
       }
       ```

       NOTE: The simplest UX per CONTEXT.md decision #11 is "Ver más reseñas" replaces visible reviews with next page (NOT infinite append). This is acceptable for ~10-50 total reviews. The hook above implements page replacement. If executor prefers append, switch to `useInfiniteQuery` from TanStack — but page replacement keeps it simpler and matches the small-volume use case.
