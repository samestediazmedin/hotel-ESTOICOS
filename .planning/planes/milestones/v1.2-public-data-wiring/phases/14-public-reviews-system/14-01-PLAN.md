       - `signReviewToken({reservationId, guestName, stayDate}): {token, jti}` — uses `jwtService.sign(payload, {expiresIn:'90d', secret: process.env.REVIEW_TOKEN_SECRET ?? process.env.JWT_ACCESS_SECRET})`, `jti = randomUUID()` from `node:crypto`

    3. **Create `reviews-public.controller.ts`** (`@Controller('public')`):
       ```typescript
       @Controller('public')
       @UseGuards(ThrottlerGuard)
       export class ReviewsPublicController {
         constructor(private readonly reviewsService: ReviewsService) {}

         @Post('reviews')
         @Throttle({ 'reviews-submit': { limit: 5, ttl: 3_600_000 } })
         async submit(@Body() body: unknown) {
           const dto = submitReviewSchema.parse(body);
           const review = await this.reviewsService.submitReview(dto);
           return { id: review.id, createdAt: review.createdAt };
         }

         @Get('reviews/validate-token')
         async validateToken(@Query('token') token: string) {
           if (!token) throw new BadRequestException('token required');
           return this.reviewsService.validateToken(token);
         }

         @Get('reviews')
         @Header('Cache-Control', 'public, max-age=60')
         async list(@Query() query: unknown) {
           const dto = publicReviewsQuerySchema.parse(query);
           return this.reviewsService.getPublicReviews(dto);
         }
       }
       ```
       NOTE: `validate-token` route registered BEFORE any `:id` route would be (P7 pitfall).
